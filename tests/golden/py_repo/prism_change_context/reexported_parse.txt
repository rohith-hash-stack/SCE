```python
def parse(text):
    """Parse text into a list of records; blank text gives an empty list."""
    if text is None:
        raise ValueError("text is required")
    if not text.strip():
        return []
    return [tokenize(line) for line in text.splitlines() if line.strip()]
```

```python
def render(records, width=20):
    """Render records as fixed-width lines."""
    return "\n".join(" ".join(field.ljust(width) for field in record) for record in records)
```
